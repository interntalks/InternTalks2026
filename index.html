2. Apps Script Backend Instructions

  1. Open your Google Sheet containing registered attendees.
  2. Ensure column headers exist (e.g. ID / Ticket Code and Status).
  3. Go to Extensions > Apps Script in the top menu.
  4. Replace all code in Code.gs with the snippet below.
  5. Click Deploy > New Deployment.
  6. Select type Web app, set Execute as to Me, and Who has access to Anyone.
  7. Click Deploy, copy the Web App URL, and paste it into the field above!
Google Apps Script Snippet (Code.gs) 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var targetId = String(data.id || data.ticketId || data.data).trim();
    
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var rows = sheet.getDataRange().getValues();
    
    // Find matching attendee by ID (Searches first column or any matching cell)
    var updated = false;
    for (var i = 1; i < rows.length; i++) {
      var rowId = String(rows[i][0]).trim(); // Assumes ID is Column A
      
      if (rowId === targetId || String(rows[i][1]).trim() === targetId) {
        // Change Status to 'Present' (Assumes Status is Column C / Index 3, adjust index as needed)
        // You can adjust sheet.getRange(row, col) to match your sheet column layout
        sheet.getRange(i + 1, 3).setValue("Present"); 
        sheet.getRange(i + 1, 4).setValue(new Date().toLocaleString()); // Optional check-in timestamp
        updated = true;
        break;
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ 
      success: updated, 
      message: updated ? "Status updated to Present" : "Attendee ID not found in sheet" 
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      success: false, 
      error: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
